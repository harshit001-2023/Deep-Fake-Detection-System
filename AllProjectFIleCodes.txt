def initialize_detector():

def build_model(input_shape=(5, 224, 224, 3), load_saved=True, model_path="models/deepfake_detector.h5"):
    """
    Builds or loads the deepfake detection model (MobileNetV2 + LSTM).

    Args:
        input_shape (tuple): Shape of the input sequence (frames, height, width, channels).
        load_saved (bool): Whether to attempt loading a saved model.
        model_path (str): Path to save/load the model.

    Returns:
        tensorflow.keras.models.Model or None: The compiled Keras model or None if error.
    """
    try:
        if load_saved and os.path.exists(model_path):
            logger.info(f"Loading saved model from {model_path}...")
            # Load the model. If custom objects are involved and not registered,
            # loading might fail. Ensure model was saved correctly.
            model = load_model(model_path)
            logger.info("Saved model loaded successfully.")
            return model

        logger.info("Building new model...")
        # --- Frame Feature Extractor ---
        # Base model: MobileNetV2 for feature extraction
        base_model = MobileNetV2(weights='imagenet', include_top=False,
                                 input_shape=input_shape[1:], # Shape of a single frame
                                 pooling='avg', name='mobilenetv2_base') # Use avg pooling directly and name the layer

        # Initially freeze the base model layers
        base_model.trainable = False

        # Create a functional model for frame processing
        frame_input = Input(shape=input_shape[1:], name="frame_input")
        # Ensure base_model call respects training=False for frozen state
        x = base_model(frame_input, training=False)
        # Add Dense layers for feature refinement
        x = Dense(256, activation='relu', kernel_regularizer=l2(0.001), name="feature_dense_1")(x)
        x = Dropout(0.4, name="feature_dropout_1")(x) # Increased dropout
        frame_features_output = Dense(128, activation='relu', kernel_regularizer=l2(0.001), name="frame_features")(x)

        # Frame processing model
        frame_processor = Model(inputs=frame_input, outputs=frame_features_output, name="frame_processor")

        # --- Sequence Model (LSTM) ---
        sequence_input = Input(shape=input_shape, name="sequence_input")

        # Apply the frame processor to each frame in the sequence
        time_distributed_features = TimeDistributed(frame_processor, name="time_distributed")(sequence_input)

        # LSTM layer to capture temporal dependencies
        lstm_out = LSTM(128, return_sequences=False, dropout=0.3, recurrent_dropout=0.3, name="lstm_layer")(time_distributed_features) # Added recurrent dropout

        # Final classification layers
        x = Dense(64, activation='relu', name="final_dense_1")(lstm_out)
        x = Dropout(0.4, name="final_dropout_1")(x) # Increased dropout
        outputs = Dense(1, activation='sigmoid', name="output_sigmoid")(x) # Binary classification

        # --- Final Model ---
        model = Model(inputs=sequence_input, outputs=outputs, name="Deepfake_Detector_MobileNetV2_LSTM")

        logger.info("Model built successfully.")
        # Compile the model here or in the training script. Compiling here for completeness.
        # Use a default learning rate; training script can override/recompile if needed.
        model.compile(optimizer=Adam(learning_rate=1e-4),
                      loss='binary_crossentropy',
                      metrics=['accuracy'])
        logger.info("Model compiled with default Adam optimizer (lr=1e-4).")
        return model

    except Exception as e:
        logger.error(f"Error building/loading model: {e}", exc_info=True)
        return None

def fine_tune_model(model, fine_tune_at_layer_name_part='block_13', lr=1e-5):
    """
    Unfreezes layers of the base model (MobileNetV2) for fine-tuning
    and recompiles the model with a lower learning rate.

    Args:
        model (tf.keras.Model): The model to fine-tune.
        fine_tune_at_layer_name_part (str): A part of the layer name from where to start unfreezing (e.g., 'block_13').
        lr (float): Learning rate for fine-tuning.

    Returns:
        tf.keras.Model: The recompiled model ready for fine-tuning, or original if error.
    """
    try:
        # Find the base model within the TimeDistributed layer's frame_processor
        frame_processor = model.get_layer('time_distributed').layer
        base_model = frame_processor.get_layer('mobilenetv2_base') # Use the name assigned in build_model
        base_model.trainable = True
        logger.info("Base model set to trainable.")

        # Find the layer index to fine-tune from
        fine_tune_from_index = -1
        for i, layer in enumerate(base_model.layers):
            if fine_tune_at_layer_name_part in layer.name:
                fine_tune_from_index = i
                break

        if fine_tune_from_index == -1:
             logger.warning(f"Layer name part '{fine_tune_at_layer_name_part}' not found in base model. Fine-tuning all base layers.")
        else:
            # Freeze all layers before the fine_tune_from_index
            for layer in base_model.layers[:fine_tune_from_index]:
                layer.trainable = False
            logger.info(f"Unfreezing base model layers from layer {fine_tune_from_index} ('{base_model.layers[fine_tune_from_index].name}') onwards.")

        # Recompile the entire model with a lower learning rate
        model.compile(optimizer=Adam(learning_rate=lr), # Use a smaller LR for fine-tuning
                      loss='binary_crossentropy',
                      metrics=['accuracy'])
        logger.info(f"Model recompiled for fine-tuning with learning rate {lr}.")
        return model
    except Exception as e:
        logger.error(f"Error during model fine-tuning setup: {e}", exc_info=True)
        # Return the original model (possibly uncompiled or compiled with previous settings)
        return model

@tf.function # Decorate with tf.function for potential performance improvement
def augment_frame(image):
    """Applies random augmentations to a single image tensor (expects values in [0, 1])."""
    image = tf.image.random_flip_left_right(image)
    image = tf.image.random_brightness(image, max_delta=0.15) # Slightly reduced delta
    # image = tf.image.random_contrast(image, lower=0.85, upper=1.15) # Optional: Add contrast
    # Consider adding slight rotation or zoom if needed, but be careful with faces
    # image = tf.contrib.image.rotate(image, tf.random.uniform((), minval=-0.1, maxval=0.1)) # Example rotation (requires tf.contrib or tf-addons)
    image = tf.clip_by_value(image, 0.0, 1.0) # Ensure values stay in [0, 1]
    return image

def process_video(video_path, num_frames=5, augment=False, target_size=(224, 224)):
    """
    Extracts, processes, and potentially augments face frames from a video.

    Args:
        video_path (str): Path to the video file.
        num_frames (int): Number of frames to extract.
        augment (bool): Whether to apply augmentation to the extracted frames.
        target_size (tuple): The target size (height, width) for the face crop.

    Returns:
        np.ndarray or None: Array of processed frames (num_frames, height, width, 3) or None if error.
    """
    if not initialize_detector():
        logger.error("Cannot process video, MTCNN detector failed to initialize.")
        return None

    cap = None # Initialize cap to None
    try:
        cap = cv2.VideoCapture(video_path)
        if not cap.isOpened():
            logger.error(f"Could not open video: {video_path}")
            return None

        total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
        # Ensure video has at least num_frames frames
        if total_frames < num_frames:
            logger.warning(f"Video '{os.path.basename(video_path)}' too short ({total_frames} frames, need {num_frames}). Skipping.")
            cap.release()
            return None

        processed_frames = []
        # Select evenly spaced frame indices
        frame_indices = np.linspace(0, total_frames - 1, num_frames, dtype=int)
        logger.debug(f"Processing frames at indices {frame_indices} for video {os.path.basename(video_path)}")

        for idx in frame_indices:
            cap.set(cv2.CAP_PROP_POS_FRAMES, idx)
            ret, frame = cap.read()
            if not ret:
                logger.warning(f"Could not read frame {idx} from video {os.path.basename(video_path)}")
                continue # Try to get the required number of frames even if some fail

            try:
                frame_rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                # Use MTCNN to detect faces
                detections = detector.detect_faces(frame_rgb)

                if detections:
                    # Find the face with the largest bounding box area
                    largest_detection = max(detections, key=lambda det: det['box'][2] * det['box'][3])
                    x, y, w, h = largest_detection['box']

                    # Ensure coordinates are valid and add slight padding
                    pad_x = int(w * 0.1) # 10% padding horizontally
                    pad_y = int(h * 0.1) # 10% padding vertically
                    x1 = max(0, x - pad_x)
                    y1 = max(0, y - pad_y)
                    x2 = min(frame_rgb.shape[1], x + w + pad_x)
                    y2 = min(frame_rgb.shape[0], y + h + pad_y)

                    face = frame_rgb[y1:y2, x1:x2]

                    if face.size == 0:
                         logger.warning(f"Empty face crop obtained for frame {idx} in {os.path.basename(video_path)} after padding.")
                         continue

                    # Resize face to the target size
                    face_resized = cv2.resize(face, (target_size[1], target_size[0]), interpolation=cv2.INTER_AREA)
                    # Normalize pixel values to [0, 1]
                    face_normalized = face_resized.astype('float32') / 255.0

                    processed_frames.append(face_normalized)
                else:
                    logger.warning(f"No face detected in frame {idx} of video {os.path.basename(video_path)}")
                    # If no face is detected, we might need to skip the video or append a placeholder.
                    # Skipping the video if we can't get enough frames is safer.

            except Exception as e:
                logger.error(f"Error processing frame {idx} from {os.path.basename(video_path)}: {e}", exc_info=True)
                continue # Continue to the next frame index

        cap.release() # Ensure cap is released

        # Check if we collected enough valid frames
        if len(processed_frames) == num_frames:
            frames_array = np.array(processed_frames)
            # Apply augmentation if requested (only during training)
            if augment:
                # Apply augmentation frame by frame
                augmented_frames = [augment_frame(tf.convert_to_tensor(f)).numpy() for f in frames_array]
                logger.debug(f"Augmented frames for {os.path.basename(video_path)}")
                return np.array(augmented_frames)
            else:
                return frames_array
        else:
            # Log if not enough frames were successfully processed
            logger.warning(f"Could not extract required {num_frames} face frames from {os.path.basename(video_path)} (extracted: {len(processed_frames)}). Skipping video.")
            return None

    except Exception as e:
        logger.error(f"General error processing video {video_path}: {e}", exc_info=True)
        # Ensure capture is released in case of general error
        if cap and cap.isOpened():
            cap.release()
        return None


def load_videos(folder_path, label, num_frames=5, max_videos=None, augment=False, target_size=(224, 224)):
    """
    Loads and processes videos from a folder.

    Args:
        folder_path (str): Path to the folder containing videos.
        label (int): Label to assign (e.g., 0 for fake, 1 for real).
        num_frames (int): Number of frames to extract per video.
        max_videos (int, optional): Maximum number of videos to load from this folder. Defaults to None (load all).
        augment (bool): Whether to apply augmentation (should be True only for training data).
        target_size (tuple): Target size for face crops.

    Returns:
        tuple: (list of processed video frame sequences, list of corresponding labels)
    """
    videos_data = []
    labels = []

    try:
        if not os.path.exists(folder_path):
            logger.error(f"Dataset path does not exist: {folder_path}")
            return videos_data, labels

        # List all supported video files
        video_files = [f for f in os.listdir(folder_path)
                       if f.lower().endswith(('.mp4', '.avi', '.mov', '.mkv'))]

        if not video_files:
            logger.warning(f"No supported video files found in {folder_path}")
            return videos_data, labels

        logger.info(f"Found {len(video_files)} videos in {folder_path}. Processing...")
        processed_count = 0

        # Limit the number of videos if max_videos is set
        files_to_process = video_files[:max_videos] if max_videos is not None else video_files

        for video_file in files_to_process:
            video_path = os.path.join(folder_path, video_file)
            # Process video with specified parameters
            processed_frames = process_video(video_path, num_frames=num_frames, augment=augment, target_size=target_size)

            if processed_frames is not None:
                videos_data.append(processed_frames)
                labels.append(label)
                processed_count += 1
                # Log progress periodically
                if processed_count % 50 == 0: # Log every 50 videos
                     logger.info(f"Processed {processed_count}/{len(files_to_process)} videos from {os.path.basename(folder_path)}...")

        logger.info(f"Successfully loaded and processed {len(videos_data)} videos from {folder_path}")
        return videos_data, labels

    except Exception as e:
        logger.error(f"Error loading videos from {folder_path}: {e}", exc_info=True)
        # Return whatever was processed successfully before the error
        return videos_data, labels

def predict_deepfake(video_path, model, num_frames=5, target_size=(224, 224)):
    """
    Predicts if a video is real or fake using the loaded model.

    Args:
        video_path (str): Path to the video file.
        model (tf.keras.Model): The trained deepfake detection model.
        num_frames (int): Number of frames to process for prediction.
        target_size (tuple): Target size for face crops.

    Returns:
        tuple: (result_string, confidence_percentage) or ("Error", 0)
    """
    try:
        if not os.path.exists(video_path):
            logger.error(f"Video file not found: {video_path}")
            return "Error: Video file not found", 0

        if model is None:
            logger.error("Model not loaded or provided for prediction.")
            return "Error: Model not available", 0

        # Process the video without augmentation for prediction
        frames_sequence = process_video(video_path, num_frames=num_frames, augment=False, target_size=target_size)

        if frames_sequence is None:
            # Error message already logged in process_video
            return "Error: Could not process video (check logs)", 0

        # Ensure the input shape matches the model's expected input
        # Add the batch dimension: (num_frames, H, W, C) -> (1, num_frames, H, W, C)
        sequence_batch = np.expand_dims(frames_sequence, axis=0)

        # Make prediction
        prediction = model.predict(sequence_batch, verbose=0)[0][0] # Get the scalar prediction value
        logger.info(f"Raw prediction score for {os.path.basename(video_path)}: {prediction:.4f}")

        # Interpret prediction using a standard threshold
        threshold = 0.5
        is_fake = prediction < threshold
        result = "Fake" if is_fake else "Real"

        # Calculate confidence (how far the prediction is from the threshold, scaled to 100)
        # If fake (pred < 0.5), confidence is based on (0.5 - pred) / 0.5 = 1 - 2*pred -> (1-pred) is simpler approx
        # If real (pred >= 0.5), confidence is based on (pred - 0.5) / 0.5 = 2*pred - 1 -> pred is simpler approx
        # Using simpler confidence: probability of the predicted class
        confidence = (1 - prediction) if is_fake else prediction
        confidence_percentage = float(confidence * 100)

        logger.info(f"Prediction for {os.path.basename(video_path)}: {result} with {confidence_percentage:.2f}% confidence")
        return result, confidence_percentage

    except Exception as e:
        logger.error(f"Error during prediction for {video_path}: {e}", exc_info=True)
        return "Error: Prediction failed unexpectedly", 0

def load_all_data(augment_train=True):

        return None, None # Indicate error

    # Log dataset composition
    num_real = np.sum(np.array(all_labels) == 1)
    num_fake = np.sum(np.array(all_labels) == 0)

    X = np.array(all_videos, dtype=np.float32) # Ensure float32
    y = np.array(all_labels, dtype=np.int32)   # Ensure int32

def train_deepfake_detector():
    """Loads data, builds/loads model, trains, fine-tunes, and saves results."""
    logger.info("===== Starting DeepFake Detector Training =====")
    start_time = time.time()

    # Ensure models directory exists for saving outputs
    os.makedirs(MODEL_DIR, exist_ok=True)

    # 1. Load Data
    # Load data with augmentation enabled for the training set implicitly
    X, y = load_all_data(augment_train=True)
    if X is None or y is None or len(X) == 0:
        logger.error("Training cannot proceed due to data loading issues.")
        return False

    # 2. Split Data (Train/Validation)
    try:
        X_train, X_val, y_train, y_val = train_test_split(
            X, y,
            test_size=VALIDATION_SPLIT,
            random_state=42,  # For reproducibility
            stratify=y      # Ensure similar class distribution in train/val splits
        )
        logger.info(f"Data split: Training set size: {len(X_train)}, Validation set size: {len(X_val)}")
    except ValueError as e:
         logger.error(f"Error during train/test split: {e}. Check if dataset size is sufficient for stratification (needs >=2 samples per class).")
         logger.info(f"Dataset shape: X={X.shape}, y={y.shape}")
         return False

    # 3. Calculate Class Weights (to handle potential imbalance)
    try:
        class_weights = class_weight.compute_class_weight(
            'balanced',
            classes=np.unique(y_train),
            y=y_train
        )
        # Create a dictionary mapping class indices (0, 1) to their weights
        class_weight_dict = dict(enumerate(class_weights))
        logger.info(f"Calculated class weights for training: {class_weight_dict}")
    except Exception as e:
        logger.error(f"Could not compute class weights: {e}. Proceeding without class weighting.")
        class_weight_dict = None # Set to None if calculation fails

    # 4. Build or Load Model
    # Attempt to load checkpoint first, otherwise build new
    latest_checkpoint = tf.train.latest_checkpoint(MODEL_DIR)
    if latest_checkpoint:
        logger.info(f"Loading model from latest checkpoint: {latest_checkpoint}")
        # Build model structure first, then load weights
        model = build_model(input_shape=(NUM_FRAMES, TARGET_SIZE[0], TARGET_SIZE[1], 3), load_saved=False) # Build structure
        if model:
            model.load_weights(latest_checkpoint) # Load weights from checkpoint
            logger.info("Model weights loaded successfully from checkpoint.")
        else:
             logger.error("Failed to build model structure. Cannot load checkpoint.")
             return False
    else:

        batch_size=BATCH_SIZE,
        validation_data=(X_val, y_val),
        callbacks=callbacks_initial,
        class_weight=class_weight_dict, # Apply class weights if available
        verbose=1
    )
    logger.info("--- Initial Training Phase Completed ---")

    # It's generally best practice to load the best weights saved by the checkpoint callback
    # before proceeding to fine-tuning, even though restore_best_weights=True is set.
    try:
        # Find the best checkpoint based on the filename pattern (requires parsing or finding latest)
        # Simplified: Assume the last checkpoint saved by ModelCheckpoint is the best one if restore_best_weights worked.
        # Or load the explicitly saved best model if ModelCheckpoint saved the full model.
        best_model_path = CHECKPOINT_PATH.format(epoch=np.argmax(history_initial.history['val_accuracy']) + 1, val_accuracy=np.max(history_initial.history['val_accuracy']))

        # Check if the best model path exists (if ModelCheckpoint saved the full model)
        if os.path.exists(best_model_path):
             logger.info(f"Loading best weights from initial phase checkpoint: {best_model_path}")
             # Since ModelCheckpoint saves the full model, we can load it directly
             model = load_model(best_model_path) # Reload the best model
             if model is None:
                 logger.error("Failed to load the best model from checkpoint. Proceeding with weights from end of initial training.")
                 # Fallback: model already has weights from end of training due to fit()
             else:
                  logger.info("Best model loaded successfully.")

        elif tf.train.latest_checkpoint(MODEL_DIR): # Fallback if only weights were saved or naming is different
            latest_chkp = tf.train.latest_checkpoint(MODEL_DIR)
            logger.warning(f"Best model file not found at expected path. Loading weights from latest checkpoint: {latest_chkp}")
            model.load_weights(latest_chkp)

        else:
            logger.warning("No checkpoint files found. Proceeding with weights from the end of initial training.")
            # model already has weights from end of training
    except Exception as e:
        logger.error(f"Error loading best weights after initial training: {e}. Proceeding with current weights.", exc_info=True)

    # --- Fine-Tuning Phase (Base Model Partially Unfrozen) ---
    logger.info(f"--- Starting Fine-Tuning Phase ({FINE_TUNE_EPOCHS} epochs, LR={FINE_TUNE_LR}) ---")

    # Unfreeze layers and recompile for fine-tuning
    model = fine_tune_model(model, fine_tune_at_layer_name_part=FINE_TUNE_LAYER_NAME_PART, lr=FINE_TUNE_LR)
    if model is None: # Check if fine_tune_model itself failed
         logger.error("Failed to prepare model for fine-tuning.")
         # Consider saving the model from the initial phase before exiting
         # model.save(MODEL_PATH.replace('.h5', '_initial.h5'))
         return False

    # Log model structure again after unfreezing layers
    model.summary(print_fn=logger.info)

    # Define callbacks for fine-tuning (can reuse or redefine)
    # Use slightly more patience for fine-tuning potentially
    early_stopping_ft = EarlyStopping(monitor='val_loss', patience=7, restore_best_weights=True, verbose=1)
    reduce_lr_ft = ReduceLROnPlateau(monitor='val_loss', factor=0.2, patience=4, min_lr=1e-7, verbose=1) # Allow even lower LR
    # Update checkpoint path for fine-tuning phase to avoid overwriting initial best
    checkpoint_path_ft = CHECKPOINT_PATH.replace('.h5', '_ft.h5')
    model_checkpoint_ft = ModelCheckpoint(checkpoint_path_ft, monitor='val_accuracy', save_best_only=True, save_weights_only=False, verbose=1, mode='max')

    callbacks_finetune = [early_stopping_ft, reduce_lr_ft, model_checkpoint_ft]

    # Calculate the total number of epochs completed so far
    epochs_done_initial = len(history_initial.epoch)

    # Continue training (fine-tuning)
    history_finetune = model.fit(
        X_train, y_train,
        epochs=epochs_done_initial + FINE_TUNE_EPOCHS, # Total epochs target
        initial_epoch=epochs_done_initial, # Start from where initial training left off
        batch_size=BATCH_SIZE, # Can potentially use smaller batch size for fine-tuning if memory constrained
        validation_data=(X_val, y_val),
        callbacks=callbacks_finetune,
        class_weight=class_weight_dict, # Continue using class weights
        verbose=1
    )
    logger.info("--- Fine-Tuning Phase Completed ---")

    # 7. Save Final Model
    # The best weights should have been restored by EarlyStopping or loaded from the best fine-tuning checkpoint.
    # It's usually best to save the model explicitly after training concludes.
    logger.info(f"Saving final trained model to {MODEL_PATH}")
    model.save(MODEL_PATH)
    logger.info("Final model saved successfully.")

    # 8. Save Combined Training History
    # Combine history dictionaries from both phases
    combined_history = {}
    for key in history_initial.history.keys():
        # Ensure keys exist in both histories before combining
        if key in history_finetune.history:
             combined_history[key] = history_initial.history[key] + history_finetune.history[key]
        else:
             combined_history[key] = history_initial.history[key] # Only include initial if fine-tuning history missing key

    # Add keys present only in fine-tuning history (less likely but possible)
    for key in history_finetune.history.keys():
        if key not in combined_history:
            # Pad with zeros or NaNs for initial phase if needed, or just start from fine-tuning
             combined_history[key] = ([0.0] * epochs_done_initial) + history_finetune.history[key] # Example padding


    # Ensure all history values are JSON serializable (convert numpy types to float)
    serializable_history = {}
    for key, value in combined_history.items():
        try:
            # Attempt conversion to list of floats
            serializable_history[key] = [float(v) for v in value]
        except (TypeError, ValueError) as e:
             logger.warning(f"Could not serialize history for key '{key}': {e}. Skipping this key.")


    if serializable_history: # Only save if we have valid data
        try:
            with open(HISTORY_PATH, 'w') as f:
                json.dump(serializable_history, f, indent=4) # Use indent for readability
            logger.info(f"Combined training history saved to {HISTORY_PATH}")
            # logger.debug(f"Combined History Data: {json.dumps(serializable_history, indent=4)}") # Optional: Log history data
        except Exception as e:
            logger.error(f"Failed to save training history: {e}", exc_info=True)
    else:
        logger.error("No valid training history data was generated or serialized. History file not saved.")


    end_time = time.time()
    total_training_time = end_time - start_time
    logger.info(f"===== Training Completed in {time.strftime('%H:%M:%S', time.gmtime(total_training_time))} =====")

    # Evaluate final model on validation set (using the model state after training/restoring best weights)
    try:
        loss, accuracy = model.evaluate(X_val, y_val, verbose=0)
        logger.info(f"Final Validation Accuracy: {accuracy*100:.2f}%")
        logger.info(f"Final Validation Loss: {loss:.4f}")
    except Exception as e:
        logger.error(f"Error evaluating final model on validation set: {e}", exc_info=True)


    return True

def to_base64(image):
    """Converts an OpenCV image (numpy array BGR) to a base64 string for HTML display."""
    try:
        if isinstance(image, np.ndarray):
            ret, buffer = cv2.imencode('.jpg', image) # Encode as JPEG
            if ret:
                return base64.b64encode(buffer).decode('utf-8')
        logger.warning("Invalid image type passed to to_base64.")
        return ''
    except Exception as e:
        logger.error(f"Error in to_base64: {e}", exc_info=True)
        return ''

@app.route('/train', methods=['POST'])
def contribute_and_train():
    """Handles upload of training videos and triggers retraining."""
    global training_in_progress
    if training_in_progress:
         # Return specific status code for "too many requests" or "service unavailable"
         return jsonify({'success': False, 'error': 'Training is already in progress. Please wait.'}), 429

    # --- Input Validation ---
    if 'file' not in request.files: return jsonify({'success': False, 'error': 'No file part'}), 400
    file = request.files['file']
    label = request.form.get('label')
    if not file or file.filename == '': return jsonify({'success': False, 'error': 'No file selected'}), 400
    if label not in ['real', 'fake']: return jsonify({'success': False, 'error': 'Invalid label specified'}), 400
    if not file.filename.lower().endswith(('.mp4', '.avi', '.mov', '.mkv')): return jsonify({'success': False, 'error': 'Invalid file format'}), 400

    try:
        # Determine target directory based on label
        target_dir = REAL_VIDEOS_PATH if label == 'real' else FAKE_VIDEOS_PATH
        os.makedirs(target_dir, exist_ok=True) # Ensure directory exists

        # Create secure filename for contributed video
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        base, ext = os.path.splitext(file.filename)
        safe_base = "".join(c if c.isalnum() else "_" for c in base)[:50]
        secure_filename = f"contrib_{timestamp}_{safe_base}{ext}"
        video_path = os.path.join(target_dir, secure_filename)

        file.save(video_path)
        logger.info(f"Training video contributed: {video_path}, Label: {label}")

        # --- Trigger Retraining Logic ---
        stats = get_dataset_stats()
        retrain_threshold = 5 # Retrain every N new videos contributed
        total_videos = stats.get('total_videos', 0)
        # Trigger if threshold met AND not already training
        if total_videos > 0 and total_videos % retrain_threshold == 0:
            logger.info(f"Dataset size ({total_videos}) reached threshold ({retrain_threshold}). Triggering retraining...")
            training_in_progress = True # Set flag BEFORE starting thread
            # Run training script in a background thread
            thread = Thread(target=run_training_script, daemon=True) # Use daemon thread
            thread.start()
            return jsonify({'success': True, 'message': f'Video contributed. Retraining started in background.'})
        else:
            return jsonify({'success': True, 'message': f'Video contributed successfully. Total videos: {total_videos}.'})

    except Exception as e:
        logger.error(f"Error saving contributed training video: {e}", exc_info=True)
        # Return a server error status code
        return jsonify({'success': False, 'error': f'An internal error occurred: {e}'}), 500

def run_training_script():
    """Runs the train.py script as a subprocess and reloads the model."""
    global model, training_in_progress
    logger.info("--- Background Training Thread Started ---")
    try:
        python_executable = sys.executable # Get path to current python interpreter
        script_path = os.path.abspath("train.py") # Get absolute path to train.py

        if not os.path.exists(script_path):
             logger.error(f"Training script not found at {script_path}")
             # Reset flag even if script not found
             training_in_progress = False
             return

        logger.info(f"Executing training script: {python_executable} {script_path}")
        # Execute train.py using subprocess, capture output
        process = subprocess.Popen([python_executable, script_path], stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True, encoding='utf-8')
        stdout, stderr = process.communicate() # Wait for completion

        # Log output from the training script
        logger.info("--- Training Script Output ---")
        if stdout: logger.info("stdout:\n" + stdout)
        if stderr: logger.error("stderr:\n" + stderr) # Log stderr as error
        logger.info("--- End Training Script Output ---")

        # Check return code and reload model
        if process.returncode == 0:
            logger.info("Training script completed successfully. Reloading model...")
            # Attempt to reload the model
            if not load_model():
                 logger.error("Failed to reload model after successful training.")
            else:
                 logger.info("Model reloaded successfully after training.")
        else:
            logger.error(f"Training script failed with return code {process.returncode}.")
            # Model is not reloaded on failure

    except Exception as e:
        logger.error(f"Error running training script in background: {e}", exc_info=True)
    finally:
        training_in_progress = False # Reset flag regardless of outcome
        logger.info("--- Background Training Thread Finished ---")

@app.route('/video_feed')
def video_feed():
    """Streams the uploaded video back to the page."""
    video_path = session.get('video_path')
    if not video_path or not os.path.exists(video_path):
        logger.warning("Video feed requested but no valid video path in session.")
        # Return a 404 Not Found status code
        return "No video loaded", 404
    # Stream frames using a generator function
    return Response(generate_frames(video_path), mimetype='multipart/x-mixed-replace; boundary=frame')

def generate_frames(video_path):
    """Generator function to read video frames and yield them for streaming."""
    cap = None # Ensure cap is defined in the outer scope
    try:
        cap = cv2.VideoCapture(video_path)
        if not cap.isOpened():
            logger.error(f"Could not open video for streaming: {video_path}")
            return # Exit generator if video can't be opened

        logger.debug(f"Starting video stream for {video_path}")
        while True:
            ret, frame = cap.read()
            if not ret:
                logger.debug(f"End of video stream reached for {video_path}")
                break # End of video

            # Encode frame as JPEG
            ret_encode, buffer = cv2.imencode('.jpg', frame, [int(cv2.IMWRITE_JPEG_QUALITY), 80]) # Adjust quality if needed
            if not ret_encode:
                logger.warning("Could not encode frame to JPEG during streaming.")
                continue # Skip this frame

            # Yield the frame in the multipart format
            frame_bytes = buffer.tobytes()
            yield (b'--frame\r\n'
                   b'Content-Type: image/jpeg\r\n\r\n' + frame_bytes + b'\r\n')
            # time.sleep(0.01) # Optional small delay for stability, adjust as needed

    except Exception as e:
         # Log error during frame generation
         logger.error(f"Error during video frame generation for {video_path}: {e}", exc_info=True)
    finally:
        # Ensure the video capture is released
        if cap and cap.isOpened():
            cap.release()
        logger.debug(f"Video stream generator finished for {video_path}")

def load_training_history():
    """Loads training history from the JSON file."""
    try:
        if os.path.exists(HISTORY_PATH):
            with open(HISTORY_PATH, 'r') as f:
                history = json.load(f)
                # Validate basic structure
                if isinstance(history, dict) and all(k in history and isinstance(history[k], list) for k in ['accuracy', 'val_accuracy', 'loss', 'val_loss']):
                    logger.debug(f"Loaded training history from {HISTORY_PATH}")
                    return history
                else:
                    logger.warning(f"Invalid format in {HISTORY_PATH}. Missing keys or not lists.")
                    return None
        else:
            logger.warning(f"Training history file not found at {HISTORY_PATH}")
            return None
    except json.JSONDecodeError:
        logger.error(f"Error decoding JSON from {HISTORY_PATH}.")
        return None
    except Exception as e:
        logger.error(f"Error loading training history: {e}", exc_info=True)
        return None

def create_training_history_graph():
    """Creates a Plotly graph dictionary for training history."""
    history = load_training_history()
    if history is None or not history.get('accuracy'):
        logger.info("No valid training history data available to plot.")
        return None
    try:
        epochs = list(range(1, len(history['accuracy']) + 1))
        # Ensure validation data exists and has same length
        if 'val_accuracy' not in history or len(epochs) != len(history['val_accuracy']):
             logger.warning("Training history 'val_accuracy' missing or has different length. Plotting only training accuracy.")
             df = pd.DataFrame({'Epoch': epochs, 'Training Accuracy': history['accuracy']})
             y_vars = ['Training Accuracy']
        else:
             df = pd.DataFrame({'Epoch': epochs, 'Training Accuracy': history['accuracy'], 'Validation Accuracy': history['val_accuracy']})
             y_vars = ['Training Accuracy', 'Validation Accuracy']

        fig = px.line(df, x='Epoch', y=y_vars,
                      title='Model Accuracy during Training',
                      labels={'value': 'Accuracy', 'variable': 'Metric'}, markers=True)
        fig.update_layout(plot_bgcolor='white', paper_bgcolor='white', font=dict(color='black'), legend_title_text='Metrics')
        fig.update_traces(line=dict(width=2))
        # Return the figure dictionary directly for JSON serialization by Flask/Jinja
        return json.loads(json.dumps(fig, cls=plotly.utils.PlotlyJSONEncoder))
    except Exception as e:
        logger.error(f"Error creating training history graph: {e}", exc_info=True)
        return None
